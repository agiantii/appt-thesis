% er_diagram.tex  —— 单独存放ER图
\begin{tikzpicture}[
    entity/.style={rectangle, draw, rounded corners, fill=blue!10, align=center, 
                  minimum width=3.2cm, minimum height=1.5cm, text width=2.9cm, inner sep=0.2cm},
    relationship/.style={rectangle, draw, fill=green!10, align=center, 
                        minimum width=1.8cm, minimum height=0.8cm, inner sep=0.1cm},
    arrow/.style={->, >=stealth, thick},
    label/.style={font=\small},
    scale=0.75, transform shape
]

% 实体节点 - 上层
\node[entity] (user) at (0, 4.8) {用户表\\users};
\node[entity, right=5.5cm of user] (space) {幻灯片空间表\\slide\_spaces};

% 实体节点 - 中层
\node[entity, below=4.2cm of user] (snippet) {代码片段表\\snippets};
\node[entity, below=4.2cm of space] (slide) {幻灯片表\\slides};
\node[entity, right=5.5cm of slide] (comment) {评论表\\slide\_comments};

% 实体节点 - 下层
\node[entity, below=4.2cm of snippet] (version) {版本控制表\\slide\_versions};
\node[entity, below=4.2cm of slide] (role) {用户角色表\\slide\_user\_roles};
\node[entity, below=4.2cm of comment] (regcode) {注册码表\\registration\_codes};

% 关系节点
\node[relationship] (user_space) at ($(user)!0.5!(space) + (0,-0.25cm)$) {拥有};
\node[relationship] (space_slide) at ($(space)!0.5!(slide) + (0.25cm,0)$) {包含};
\node[relationship] slide_version) at ($(slide)!0.5!(version) + (0.25cm,0)$) {版本化};
\node[relationship] (user_slide) at ($(user)!0.5!(slide) + (-0.15cm,-0.15cm)$) {协作};
\node[relationship] (slide_comment) at ($(slide)!0.5!(comment) + (0,0.25cm)$) {评论};
\node[relationship] (user_role) at ($(user)!0.5!(role) + (-0.25cm,0)$) {授权};

% 连接线
\draw[arrow] (user) -- node[pos=0.3, above, label] {1} (user_space);
\draw[arrow] (user_space) -- node[pos=0.7, above, label] {N} (space);

\draw[arrow] (space) -- node[pos=0.3, right, label] {1} (space_slide);
\draw[arrow] (space_slide) -- node[pos=0.7, right, label] {N} (slide);

\draw[arrow] (slide) -- node[pos=0.3, right, label] {1} (slide_version);
\draw[arrow] (slide_version) -- node[pos=0.7, right, label] {N} (version);

\draw[arrow] (user) -- node[pos=0.3, left, label] {1} (user_slide);
\draw[arrow] (user_slide) -- node[pos=0.7, above, label] {N} (slide);

\draw[arrow] (slide) -- node[pos=0.3, above, label] {1} (slide_comment);
\draw[arrow] (slide_comment) -- node[pos=0.7, above, label] {N} (comment);

\draw[arrow] (user) -- node[pos=0.3, left, label] {1} (user_role);
\draw[arrow] (user_role) -- node[pos=0.7, right, label] {N} (role);

% 其他实体连接
\draw[arrow, dashed] (user) -- node[pos=0.3, left, label] {创建} (snippet);
\draw[arrow, dashed] (user) -- node[pos=0.7, below right, label] {使用} (regcode);

\end{tikzpicture}
